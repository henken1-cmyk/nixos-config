- [ ] **Step 3: Implement monitor parser**

Hyprlang monitor format: `output, mode, position, scale, [keyword, value]…`. The first 4 fields are positional; subsequent pairs are keyword=value extras. The exception is `output` itself which may contain commas (`desc:Foo, Bar Vendor,...`) — but Hyprland's parser only treats the FIRST comma after a `desc:` prefix specially. To handle `desc:` outputs containing commas safely, we use a hyprlang-compatible split: scan from the right for the known positional fields.

For simplicity and matching how our config actually writes monitors, use this rule: if the entry starts with `desc:`, the `output` value extends up to the comma BEFORE the mode field (which matches `\d+x\d+@\d+`). Otherwise positional split.

Add to `lib/toHyprlandLua.nix`:

```nix
  # Trim leading/trailing whitespace from a string.
  trim = s:
    let s1 = lib.removePrefix " " s;
        s2 = lib.removeSuffix " " s1;
    in if s2 == s then s2 else trim s2;

  # Coerce string to int/float/bool/string in that order.
  coerce = v:
    if v == "true" then true
    else if v == "false" then false
    else if builtins.match "-?[0-9]+" v != null then lib.toInt v
    else if builtins.match "-?[0-9]+\\.[0-9]+" v != null then
      # lib.toFloat exists in nixpkgs 24.11+; if missing, fall back to builtins.fromJSON.
      (lib.toFloat or (s: builtins.fromJSON s)) v
    else v;

  # Recognised mode tokens: WxH@Hz, "preferred", "highrr", "highres", "disable".
  isMode = p: builtins.match "([0-9]+x[0-9]+@[0-9.]+|preferred|highrr|highres|disable)" p != null;

  # Pair up [k1 v1 k2 v2 ...] -> { k1 = v1; k2 = v2; }. Drops a trailing unpaired item.
  pairUp = xs:
    if builtins.length xs < 2 then {}
    else { "${builtins.head xs}" = coerce (builtins.elemAt xs 1); }
         // pairUp (lib.drop 2 xs);

  # Parse a hyprlang monitor entry into { output, mode, position, scale, ...extras }.
  # output may contain commas (desc: prefix), so we locate the mode token and
  # treat everything before it as part of the output.
  parseMonitor = entry:
    let
      parts = map trim (lib.splitString "," entry);
      modeIdx = lib.foldl' (acc: i:
        if acc != null then acc
        else if isMode (builtins.elemAt parts i) then i
        else null
      ) null (lib.range 0 (builtins.length parts - 1));
      _ = if modeIdx == null
          then throw "toHyprlandLua.parseMonitor: no mode token found in '${entry}'"
          else null;
      output = lib.concatStringsSep "," (lib.take modeIdx parts);
      rest = lib.drop modeIdx parts;
      mode = builtins.elemAt rest 0;
      position = builtins.elemAt rest 1;
      scale = builtins.elemAt rest 2;
      extras = pairUp (lib.drop 3 rest);
    in
    { inherit output mode position scale; } // extras;

  emitMonitor = entries:
    lib.concatMapStringsSep "\n" (entry:
      let m = parseMonitor entry;
          # Preserve key order: positional first, extras after, alphabetical within extras.
          positionalKeys = [ "output" "mode" "position" "scale" ];
          extraKeys = lib.sort lib.lessThan
            (lib.filter (k: !(builtins.elem k positionalKeys)) (builtins.attrNames m));
          renderKv = k: "${k} = ${lib.generators.toLua {} m.${k}}";
          body = lib.concatStringsSep ", " (map renderKv (positionalKeys ++ extraKeys));
      in "hl.monitor({ ${body} })"
    ) entries;

  monitorBlock = if attrs ? monitor then emitMonitor attrs.monitor else "";
```

The trick: when emitting the `animations` section, peel out `bezier` and `animation` keys before passing the rest to `emitSection`. Then emit `hl.curve` and `hl.animation` separately.

Add to `lib/toHyprlandLua.nix`:

```nix
  parseBezier = entry:
    let parts = map (s: lib.removePrefix " " (lib.removeSuffix " " s)) (lib.splitString "," entry);
        name = builtins.head parts;
        nums = map lib.toFloat (builtins.tail parts);
    in { inherit name; p1x = builtins.elemAt nums 0; p1y = builtins.elemAt nums 1; p2x = builtins.elemAt nums 2; p2y = builtins.elemAt nums 3; };

  emitCurve = b: ''hl.curve("${b.name}", { type = "bezier", points = { {${toString b.p1x}, ${toString b.p1y}}, {${toString b.p2x}, ${toString b.p2y}} } })'';

  parseAnimation = entry:
    let parts = map (s: lib.removePrefix " " (lib.removeSuffix " " s)) (lib.splitString "," entry);
        leaf = builtins.elemAt parts 0;
        enabled = (builtins.elemAt parts 1) == "1";
        speed = lib.toFloat (builtins.elemAt parts 2);
        bezier = builtins.elemAt parts 3;
        style = if builtins.length parts > 4 then builtins.elemAt parts 4 else null;
    in { inherit leaf enabled speed bezier style; };

  emitAnimation = a:
    let base = ''leaf = "${a.leaf}", enabled = ${lib.boolToString a.enabled}, speed = ${toString a.speed}, bezier = "${a.bezier}"'';
        withStyle = if a.style != null then ''${base}, style = "${a.style}"'' else base;
    in "hl.animation({ ${withStyle} })";

  # Override emitSection for "animations" to split out bezier/animation keys.
  emitAnimations = animSection:
    let
      bezierEntries = animSection.bezier or [];
      animEntries = animSection.animation or [];
      otherKeys = removeAttrs animSection [ "bezier" "animation" ];
      configCall = if otherKeys != {} then emitSection "animations" otherKeys else "";
      curves = lib.concatMapStringsSep "\n" (b: emitCurve (parseBezier b)) bezierEntries;
      anims = lib.concatMapStringsSep "\n" (a: emitAnimation (parseAnimation a)) animEntries;
    in lib.concatStringsSep "\n" (lib.filter (s: s != "") [ configCall curves anims ]);
```

Then in the section emission step, special-case `animations`:

```nix
  # Replace `presentSections` and `sectionBlock` logic:
  sectionParts = lib.mapAttrsToList (name: value:
    if name == "animations" then emitAnimations value
    else if builtins.elem name sectionKeys then emitSection name value
    else ""
  ) (lib.filterAttrs (k: _: builtins.elem k sectionKeys) attrs);
  sectionBlock = lib.concatStringsSep "\n" (lib.filter (s: s != "") sectionParts);
```

```nix
# returns a Lua dispatcher expression. Verified against Hyprland v0.55.0
# (src/config/lua/bindings/LuaBindingsDispatchers.cpp).

# Coverage requirement: every dispatcher appearing in our config must be here.
# Unknown dispatchers cause a generator-time error.

let
  q = s: lib.generators.toLua {} s;

  # Map hyprlang direction letters to Lua direction strings.
  dirMap = {
    "l" = "left"; "r" = "right"; "u" = "up"; "d" = "down";

in
{
  # Top-level dispatchers ---------------------------------------------------
  exec = arg: "hl.dsp.exec_cmd(${q arg})";

  killactive = _: "hl.dsp.window.close()";

  fullscreen = arg:
    let mode = if arg == "" || arg == "0" then "0"
               else if arg == "1" then "1"
               else arg;
    in "hl.dsp.window.fullscreen({ mode = ${mode} })";

  pseudo = _: "hl.dsp.window.pseudo()";

  togglefloating = _: ''hl.dsp.window.float({ action = "toggle" })'';

  movefocus = arg: ''hl.dsp.focus({ direction = ${q (parseDir arg)} })'';

  focusmonitor = arg: ''hl.dsp.focus({ monitor = ${q (parseDir arg)} })'';

  workspace = arg: ''hl.dsp.focus({ workspace = ${q arg} })'';

  movetoworkspace = arg: ''hl.dsp.window.move({ workspace = ${q arg} })'';

  movewindow = arg: ''hl.dsp.window.move({ direction = ${q (parseDir arg)} })'';

  swapwindow = arg: ''hl.dsp.window.swap({ direction = ${q (parseDir arg)} })'';

  togglespecialworkspace = arg:
    ''hl.dsp.workspace.toggle_special(${q (if arg == "" then "special" else arg)})'';

  movecurrentworkspacetomonitor = arg:
    ''hl.dsp.workspace.move({ monitor = ${q (parseDir arg)} })'';

  submap = arg: "hl.dsp.submap(${q arg})";

  layoutmsg = arg: "hl.dsp.layout(${q arg})";

}
```

```nix
  dispatchers = import ./hyprlandDispatchers.nix { inherit lib; };

  # Parse "MODS, KEY, dispatcher, arg" into structured record.
  # MODS and arg may be empty. arg may contain commas (for `exec` rest-of-line);
  # we split on all commas and rejoin parts 4+ with commas to preserve them.
  parseBind = entry:
    let
      parts = lib.splitString "," entry;
      mods = trim (builtins.elemAt parts 0);
      key = trim (builtins.elemAt parts 1);
      dispatcher = trim (builtins.elemAt parts 2);
      argParts = lib.drop 3 parts;
      arg = trim (lib.concatStringsSep "," argParts);
    in { inherit mods key dispatcher arg; };

  # Compose keychord: "SUPER, Return" → "SUPER + Return"; ", Q" → "Q".
  composeKeychord = mods: key:
    if mods == "" then key else "${mods} + ${key}";

  # Emit one bind line. flagOpts is { repeating? = true; locked? = true; mouse? = true }.
  emitBind = flagOpts: entry:
    let
      p = parseBind entry;
      dispFn = dispatchers.${p.dispatcher}
        or (throw "toHyprlandLua: unknown dispatcher '${p.dispatcher}' in bind '${entry}'. Add it to lib/hyprlandDispatchers.nix.");
      dispCall = dispFn p.arg;
      keychord = composeKeychord p.mods p.key;
      flagsBody = lib.concatStringsSep ", " (lib.mapAttrsToList
        (k: v: "${k} = ${lib.boolToString v}")
        flagOpts);
      flagsArg = if flagOpts == {} then "" else ", { ${flagsBody} }";
    in ''hl.bind(${lib.generators.toLua {} keychord}, ${dispCall}${flagsArg})'';

  emitBindList = flagOpts: entries:
    lib.concatMapStringsSep "\n" (emitBind flagOpts) entries;

  bindBlock = lib.concatStringsSep "\n" (lib.filter (s: s != "") [
    (if attrs ? bind then emitBindList {} attrs.bind else "")
    (if attrs ? binde then emitBindList { repeating = true; } attrs.binde else "")
    (if attrs ? bindl then emitBindList { locked = true; } attrs.bindl else "")
    (if attrs ? bindm then emitBindList { mouse = true; } attrs.bindm else "")
  ]);
```

```nix
  splitTrim = s: map (x: lib.removePrefix " " (lib.removeSuffix " " x)) (lib.splitString "," s);

        ) (builtins.tail parts);
        rules = builtins.listToAttrs kvPairs;
    in { inherit workspace; } // rules;

  emitWorkspace = entries:
    lib.concatMapStringsSep "\n" (e:
      let r = parseWorkspace e;
          body = lib.concatStringsSep ", " (lib.mapAttrsToList
```
